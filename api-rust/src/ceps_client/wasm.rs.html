1//! Resolve and load demo on-chain CEP contract WASM bytes from disk.
2//!
3//! Bytes live under [`ENV_WASM_ROOT`] (or discovered defaults such as
4//! `tests/wasm/`). Short names like `cep18` resolve to
5//! `{root}/cep18/cep18.wasm`. This module does not embed contract bytecode.
6
7use crate::error::{CEPError, Result};
8use std::path::{Component, Path, PathBuf};
9
10/// Environment variable for the demo contract WASM root directory.
11pub const ENV_WASM_ROOT: &str = "CEPS_WASM_ROOT";
12
13/// Resolve the demo WASM root directory.
14///
15/// Order: [`ENV_WASM_ROOT`] if set; else the first existing directory among
16/// `/opt/ceps/tests/wasm`, `tests/wasm`, `./tests/wasm`; else `tests/wasm`.
17#[must_use]
18pub fn wasm_root() -> PathBuf {
19    if let Ok(p) = std::env::var(ENV_WASM_ROOT) {
20        let trimmed = p.trim();
21        if !trimmed.is_empty() {
22            return PathBuf::from(trimmed);
23        }
24    }
25    for candidate in ["/opt/ceps/tests/wasm", "tests/wasm", "./tests/wasm"] {
26        let p = PathBuf::from(candidate);
27        if p.is_dir() {
28            return p;
29        }
30    }
31    PathBuf::from("tests/wasm")
32}

34/// Resolve `name` to a filesystem path under `root`.
35///
36/// Accepts:
37/// - empty → error
38/// - path containing `/` or `\` (absolute, or relative under `root`)
39/// - file under `root` (`cep18.wasm` or `cep18/cep18.wasm`)
40/// - short alias (`cep18` → `root/cep18/cep18.wasm`, then `root/cep18.wasm`)
41///
42/// Relative paths must not contain `..` components that escape `root`.
43///
44/// # Errors
45///
46/// Returns [`CEPError::InvalidArgument`] for an empty name or a path that
47/// escapes `root`. Returns [`CEPError::WasmNotFound`] when no candidate exists.
48pub fn resolve_path(root: &Path, name: &str) -> Result<PathBuf> {
49    let name = name.trim();
50    if name.is_empty() {
51        return Err(CEPError::InvalidArgument("wasm name is empty".into()));
52    }
53
54    let mut candidates: Vec<PathBuf> = Vec::new();
55    if name.contains('/') || name.contains('\\') {
56        let as_path = PathBuf::from(name);
57        if as_path.is_absolute() {
58            if path_has_parent(&as_path) {
59                return Err(CEPError::InvalidArgument(format!(
60                    "wasm path must not contain '..': {name}"
61                )));
62            }
63            candidates.push(as_path);
64        } else {
65            if path_escapes_root(root, &as_path) {
66                return Err(CEPError::InvalidArgument(format!(
67                    "wasm path escapes root {}: {name}",
68                    root.display()
69                )));
70            }
71            candidates.push(root.join(&as_path));
72        }
73    } else {
74        candidates.push(root.join(name));
75        if !name.ends_with(".wasm") {
76            candidates.push(root.join(format!("{name}.wasm")));
77            candidates.push(root.join(name).join(format!("{name}.wasm")));
78        }
79    }
80
81    for path in &candidates {
82        if path.is_file() {
83            return Ok(path.clone());
84        }
85    }
86
87    Err(CEPError::WasmNotFound {
88        name: name.to_string(),
89        tried: candidates.iter().map(|p| p.display().to_string()).collect(),
90    })
91}

93/// Resolve `name` under [`wasm_root`].
94///
95/// # Errors
96///
97/// Same as [`resolve_path`].
98pub fn resolve(name: &str) -> Result<PathBuf> {
99    resolve_path(&wasm_root(), name)
100}
101
102/// Read WASM bytes for `name` under `root`.
103///
104/// # Errors
105///
106/// Returns resolve errors, or [`CEPError::Io`] when the file cannot be read.
107pub fn load_from(root: &Path, name: &str) -> Result<Vec<u8>> {
108    let path = resolve_path(root, name)?;
109    std::fs::read(&path).map_err(CEPError::from)
110}
111
112/// Read WASM bytes for `name` under [`wasm_root`].
113///
114/// # Errors
115///
116/// Same as [`load_from`].
117pub fn load(name: &str) -> Result<Vec<u8>> {
118    load_from(&wasm_root(), name)
119}
120
121fn path_has_parent(path: &Path) -> bool {
122    path.components().any(|c| matches!(c, Component::ParentDir))
123}
124
125fn path_escapes_root(root: &Path, relative: &Path) -> bool {
126    if !path_has_parent(relative) {
127        return false;
128    }
129    let joined = root.join(relative);
130    let Ok(canon_root) = root.canonicalize() else {
131        // Root may not exist yet in unit tests; reject any `..` in the relative path.
132        return true;
133    };
134    match joined.canonicalize() {
135        Ok(canon) => !canon.starts_with(&canon_root),
136        Err(_) => true,
137    }
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143    use std::fs;
144    use std::io::Write;
145
146    #[test]
147    fn resolve_alias_cep18_dir() {
148        let dir = tempfile_dir();
149        let nested = dir.join("cep18");
150        fs::create_dir_all(&nested).unwrap();
151        let wasm = nested.join("cep18.wasm");
152        let mut f = fs::File::create(&wasm).unwrap();
153        writeln!(f, "fake").unwrap();
154        drop(f);
155
156        let path = resolve_path(&dir, "cep18").unwrap();
157        assert_eq!(path, wasm);
158    }
159
160    #[test]
161    fn resolve_relative_under_root() {
162        let dir = tempfile_dir();
163        let nested = dir.join("cep78");
164        fs::create_dir_all(&nested).unwrap();
165        let wasm = nested.join("mint_session.wasm");
166        fs::write(&wasm, b"x").unwrap();
167
168        let path = resolve_path(&dir, "cep78/mint_session.wasm").unwrap();
169        assert_eq!(path, wasm);
170    }
171
172    #[test]
173    fn resolve_missing_name() {
174        let dir = tempfile_dir();
175        let err = resolve_path(&dir, "cep18").unwrap_err();
176        match err {
177            CEPError::WasmNotFound { name, tried } => {
178                assert_eq!(name, "cep18");
179                assert!(!tried.is_empty());
180            }
181            other => panic!("expected WasmNotFound, got {other:?}"),
182        }
183    }
184
185    #[test]
186    fn resolve_empty_name() {
187        let dir = tempfile_dir();

190    }

192    #[test]

199    #[test]
200    fn load_from_reads_bytes() {
201        let dir = tempfile_dir();
202        let nested = dir.join("cep18");
203        fs::create_dir_all(&nested).unwrap();
204        fs::write(nested.join("cep18.wasm"), b"wasm-bytes").unwrap();
