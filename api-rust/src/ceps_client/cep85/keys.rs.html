1//! Dictionary item keys for CEP-85.
2
3use super::entity::prefixed_key;
4use crate::error::{CEPError, Result};
5use casper_rust_wasm_sdk::helpers::make_dictionary_item_key;
6use casper_rust_wasm_sdk::types::key::Key;
7use casper_types::bytesrepr::ToBytes;
8
9/// Balance dict key: blake2b(account_key ‖ id_u256) hex.
10pub fn balance_dictionary_key(account: &str, id: &str) -> Result<String> {
11    let account_key = Key::from_formatted_str(&prefixed_key(account)?)
12        .map_err(|e| CEPError::InvalidHash(format!("account key: {e}")))?;
13    let id_u256 = parse_u256(id)?;
14    Ok(make_dictionary_item_key(&account_key, &id_u256))
15}
16
17/// Operator dict key: blake2b(owner_key ‖ operator_key) hex.
18pub fn operator_dictionary_key(owner: &str, operator: &str) -> Result<String> {
19    let owner_key = Key::from_formatted_str(&prefixed_key(owner)?)
20        .map_err(|e| CEPError::InvalidHash(format!("owner key: {e}")))?;
21    let operator_key = Key::from_formatted_str(&prefixed_key(operator)?)
22        .map_err(|e| CEPError::InvalidHash(format!("operator key: {e}")))?;
23    Ok(make_dictionary_item_key(&owner_key, &operator_key))
24}
25
26/// Security badge dict item key: hex(Key.to_bytes()) under `security_badges`.
27pub fn security_badge_dictionary_key(entity: &str) -> Result<String> {
28    let key = Key::from_formatted_str(&prefixed_key(entity)?)
29        .map_err(|e| CEPError::InvalidHash(format!("badge entity key: {e}")))?;
30    let bytes = key
31        .to_bytes()
32        .map_err(|e| CEPError::InvalidHash(format!("badge entity bytes: {e}")))?;
33    Ok(hex::encode(bytes))
34}
35
36fn parse_u256(s: &str) -> Result<casper_types::U256> {
37    casper_types::U256::from_dec_str(s).map_err(|e| CEPError::InvalidArgument(format!("U256: {e}")))
38}

44    #[test]

46        let account =
47            "account-hash-b485c074cef7ccaccd0302949d2043ab7133abdb14cfa87e8392945c0bd80a5f";
48        let key = balance_dictionary_key(account, "1").unwrap();

52    #[test]

54        let account =
55            "account-hash-b485c074cef7ccaccd0302949d2043ab7133abdb14cfa87e8392945c0bd80a5f";
56        let key = security_badge_dictionary_key(account).unwrap();
57        assert!(key.chars().all(|c| c.is_ascii_hexdigit()));
58        assert!(!key.is_empty());
59        assert_eq!(
60            security_badge_dictionary_key(account).unwrap(),
61            key,
62            "stable"
