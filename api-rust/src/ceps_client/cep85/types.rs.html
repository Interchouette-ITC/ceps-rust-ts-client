5/// Arguments for CEP-85 install.
6#[derive(Debug, Clone)]
7pub struct InstallArgs {
8    /// Collection name.
9    pub name: String,
10    /// URI template.
11    pub uri: String,
12    /// Optional events mode.
13    pub events_mode: Option<EventsMode>,
14    /// Enable burn.
15    pub enable_burn: Option<bool>,
16    /// Optional admin list (entity-prefixed keys).
17    pub admin_list: Option<Vec<String>>,
18    /// Optional minter list.
19    pub minter_list: Option<Vec<String>>,
20    /// Optional burner list.
21    pub burner_list: Option<Vec<String>>,
22    /// Optional meta list.
23    pub meta_list: Option<Vec<String>>,
24}
25
26impl InstallArgs {
27    /// Required fields constructor.
28    pub fn new(name: impl Into<String>, uri: impl Into<String>) -> Self {
29        Self {
30            name: name.into(),
31            uri: uri.into(),
32            events_mode: None,
33            enable_burn: None,
34            admin_list: None,
35            minter_list: None,
36            burner_list: None,
37            meta_list: None,
38        }
39    }
40
41    /// Set events mode.
42    pub fn with_events_mode(mut self, mode: EventsMode) -> Self {
43        self.events_mode = Some(mode);
44        self
45    }
46
47    /// Enable burn.
48    pub fn with_enable_burn(mut self, enable: bool) -> Self {
49        self.enable_burn = Some(enable);
50        self
51    }
52}

54/// Arguments for CEP-85 upgrade (`upgrade: true` is always sent).
