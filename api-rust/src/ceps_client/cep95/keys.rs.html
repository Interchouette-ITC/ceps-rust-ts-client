17/// Balance dictionary item key: Base64(Key.to_bytes()) under dict `balances`.
18pub fn balance_dictionary_key(owner: &str) -> Result<String> {
19    let prefixed = prefixed_key(owner)?;
20    if prefixed.starts_with("account-hash-") {
21        get_base64_key_from_account_hash(&prefixed)
22            .map_err(|e| CEPError::InvalidHash(format!("balance key: {e}")))
23    } else {
24        get_base64_key_from_key_hash(&prefixed)
25            .map_err(|e| CEPError::InvalidHash(format!("balance key: {e}")))
26    }
27}

29/// Token-id dictionary item key (owners / approvals / token_metadata):
30/// Base64(U256.to_bytes()) as used by Odra `base64_encoded_key_value_storage`.
31pub fn token_id_dictionary_key(token_id: &str) -> Result<String> {
32    let id = U256::from_str(token_id.trim())
33        .map_err(|e| CEPError::InvalidArgument(format!("token_id U256: {e}")))?;
34    let bytes = id
35        .to_bytes()
36        .map_err(|e| CEPError::InvalidArgument(format!("token_id bytes: {e}")))?;
37    Ok(BASE64.encode(bytes))
38}

40/// Operator dictionary item key: hex(blake2b-256(owner.bytes ‖ operator.bytes)).
41pub fn operator_dictionary_key(owner: &str, operator: &str) -> Result<String> {
42    let owner_key = Key::from_formatted_str(&prefixed_key(owner)?)
43        .map_err(|e| CEPError::InvalidHash(format!("owner key: {e}")))?;
44    let operator_key = Key::from_formatted_str(&prefixed_key(operator)?)
45        .map_err(|e| CEPError::InvalidHash(format!("operator key: {e}")))?;
46    Ok(make_dictionary_item_key(&owner_key, &operator_key))
47}

49/// Odra `state` dictionary item key for a small-index Var path (all indices <= 15).
50///
51/// Odra module field indices are 1-based. Tip `OwnedCep95` stores Ownable owner at
52/// path `[ownable=1, owner=1]` → packed index `0x11`, then blake2b-256 hex (ASCII)
53/// under dict `state`.
54pub fn odra_state_var_key(path: &[u8]) -> Result<String> {
55    if path.is_empty() {
56        return Err(CEPError::InvalidArgument("odra state path empty".into()));
57    }
58    if path.iter().any(|&idx| idx > 15) {
59        return Err(CEPError::InvalidArgument(
60            "odra state path requires indices <= 15 for small encoding".into(),
61        ));
62    }
63    let index: u32 = path
64        .iter()
65        .fold(0u32, |acc, &idx| (acc << 4) + u32::from(idx));
66    let index_bytes = index.to_be_bytes();
67    let mut hasher = Blake2bVar::new(32)
68        .map_err(|_| CEPError::InvalidArgument("blake2b-256 hasher init failed".into()))?;
69    hasher.update(&index_bytes);
70    let mut hash = [0u8; 32];
71    hasher
72        .finalize_variable(&mut hash)
73        .map_err(|_| CEPError::InvalidArgument("blake2b-256 finalize failed".into()))?;
74    Ok(hex::encode(hash))
75}

77/// Ownable owner Var key for tip `OwnedCep95` (`ownable` field 1, `owner` field 1).
78pub fn ownable_owner_state_key() -> Result<String> {
79    odra_state_var_key(&[1, 1])
80}

82#[cfg(test)]

86    #[test]
87    fn balance_key_from_account_hash() {
88        let account =
89            "account-hash-b485c074cef7ccaccd0302949d2043ab7133abdb14cfa87e8392945c0bd80a5f";
90        let key = balance_dictionary_key(account).unwrap();
91        assert!(!key.is_empty());
92    }

94    #[test]
95    fn token_id_key_is_stable_base64() {
96        let a = token_id_dictionary_key("42").unwrap();
97        let b = token_id_dictionary_key("42").unwrap();
98        assert_eq!(a, b);
99        assert!(!a.is_empty());
100    }
101
102    #[test]
103    fn operator_key_is_hex() {
104        let owner = "account-hash-b485c074cef7ccaccd0302949d2043ab7133abdb14cfa87e8392945c0bd80a5f";
105        let operator =
106            "account-hash-aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa";
107        let key = operator_dictionary_key(owner, operator).unwrap();
108        assert_eq!(key.len(), 64);
109        assert!(key.chars().all(|c| c.is_ascii_hexdigit()));
110    }
111
112    #[test]
113    fn ownable_owner_state_key_stable_64_hex() {
114        let key = ownable_owner_state_key().unwrap();
115        assert_eq!(key.len(), 64);
116        assert!(key.chars().all(|c| c.is_ascii_hexdigit()));
117        assert_eq!(key, ownable_owner_state_key().unwrap());
118        // Path [1,1] → packed 0x11; must not collide with legacy wrong [0,0] key.
119        assert_ne!(key, odra_state_var_key(&[0, 0]).unwrap());
120        assert_eq!(key, odra_state_var_key(&[1, 1]).unwrap());
121    }
