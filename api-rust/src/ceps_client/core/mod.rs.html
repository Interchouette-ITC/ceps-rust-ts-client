27/// Default NCTL RPC endpoint.
28pub const DEFAULT_RPC_URL: &str = "http://127.0.0.1:11101";
29/// Default NCTL SSE events endpoint.

31/// Default NCTL chain name.

50    /// Create a core client.
51    ///
52    /// `rpc_url` is required. Empty `sse_url` is treated as unset.
53    pub fn new(
54        rpc_url: impl Into<String>,
55        sse_url: Option<String>,
56        chain_name: Option<String>,
57        verbosity: Option<Verbosity>,
58    ) -> Result<Self> {
59        let rpc_url = normalize_rpc_url(&rpc_url.into())?;
60        let sse_url = match sse_url {
61            Some(s) if !s.trim().is_empty() => Some(normalize_sse_url(&s)?),
62            _ => None,
63        };
64        let verbosity = verbosity.unwrap_or(Verbosity::Low);
65        let sdk = SDK::new(Some(rpc_url.clone()), None, Some(verbosity));
66        Ok(Self {
67            sdk,
68            rpc_url,
69            sse_url,
70            chain_name: chain_name.unwrap_or_else(|| DEFAULT_CHAIN_NAME.to_string()),
71            verbosity,
72            target: None,
73            runtime_v2: Some(false),
74            cep_kind: None,
75        })
76    }

226    /// Wait for a transaction hash on the configured SSE endpoint.
227    pub async fn wait_transaction(
228        &self,
229        transaction_hash: &str,
230        timeout_ms: Option<u64>,
231    ) -> Result<Value> {
232        let sse = self
233            .sse_url
234            .as_deref()
235            .ok_or_else(|| CepError::WaitFailed("SSE URL is not configured".into()))?;
236        let timeout = timeout_ms.or(Some(DEFAULT_WAIT_TIMEOUT_MS));
237        let event = self
238            .sdk
239            .wait_transaction(sse, transaction_hash, timeout)
240            .await
241            .map_err(CepError::WaitFailed)?;
242        serde_json::to_value(event).map_err(|e| CepError::Other(e.to_string()))
243    }

245    pub(crate) fn builder_for_entrypoint(
246        &self,
247        entry_point: &str,
248    ) -> Result<TransactionBuilderParams> {
249        let target = self.require_target()?;
250        if let Some(package_hex) = &target.package_hash {
251            let package = PackageHash::new(package_hex)
252                .map_err(|e| CepError::InvalidHash(format!("package hash: {e}")))?;
253            Ok(TransactionBuilderParams::new_package(
254                package,
255                entry_point,
256                None,
257            ))
258        } else {
259            let entity = AddressableEntityHash::new(&target.contract_hash)
260                .map_err(|e| CepError::InvalidHash(format!("contract hash: {e}")))?;
261            Ok(TransactionBuilderParams::new_invocable_entity(
262                entity,
263                entry_point,
264            ))
265        }
266    }

272    pub(crate) async fn maybe_wait(
273        &self,
274        deploy: &DeployParams,
275        mut result: CallResult,
276    ) -> Result<CallResult> {
277        if !deploy.wait {
278            return Ok(result);
279        }
280        let _event = self
281            .wait_transaction(&result.transaction_hash, deploy.wait_timeout_ms)
282            .await?;
283        let tx_hash = TransactionHash::new(&result.transaction_hash)
284            .map_err(|e| CepError::InvalidHash(format!("transaction hash: {e}")))?;
285
286        // After SSE reports processed: try to attach execution JSON when the node
287        // can return it. Some large successful installs fail NCTL deserialize.
288        let mut last_json = None;
289        for _ in 0..8 {
290            if let Ok(get_tx) = self
291                .sdk
292                .get_transaction(
293                    tx_hash.clone(),
294                    Some(false),
295                    Some(self.verbosity),
296                    Some(self.rpc_url.clone()),
297                )
298                .await
299            {
300                if let Ok(json) = serde_json::to_value(&get_tx.result) {
301                    last_json = Some(json);
302                }
303            }
304
305            #[cfg(not(target_arch = "wasm32"))]
306            if last_json.as_ref().is_none_or(|j| !json_has_execution(j)) {
307                if let Ok(json) =
308                    fetch_transaction_json_raw(&self.rpc_url, &result.transaction_hash).await
309                {
310                    last_json = Some(json);
311                }
312            }
313
314            if let Some(json) = &last_json {
315                if let Some(err) = extract_execution_error(json) {
316                    return Err(CepError::from_execution_message(err, self.cep_kind));
317                }
318                if json_has_execution(json) {
319                    break;
320                }
321            }
322            #[cfg(not(target_arch = "wasm32"))]
323            {
324                std::thread::sleep(std::time::Duration::from_millis(250));
325            }
326        }
327        if let Some(json) = last_json {
328            if let Some(err) = extract_execution_error(&json) {
329                return Err(CepError::from_execution_message(err, self.cep_kind));
330            }
331            result = result.with_execution(json);
332        }
333        Ok(result)
334    }

341fn json_has_execution(json: &Value) -> bool {
342    json.get("execution_info").is_some()
343        || json.pointer("/execution_result").is_some()
344        || json.to_string().contains("\"execution_result\"")
345}

347#[cfg(not(target_arch = "wasm32"))]
348async fn fetch_transaction_json_raw(
349    rpc_url: &str,
350    transaction_hash: &str,
351) -> std::result::Result<Value, String> {
352    let body = serde_json::json!({
353        "jsonrpc": "2.0",
354        "id": 1,
355        "method": "info_get_transaction",
356        "params": {
357            "transaction_hash": { "Version1": transaction_hash },
358            "finalized_approvals": false
359        }
360    });
361    let client = reqwest::Client::new();
362    let resp = client
363        .post(rpc_url)
364        .json(&body)
365        .send()
366        .await
367        .map_err(|e| format!("raw info_get_transaction: {e}"))?;
368    let value: Value = resp
369        .json()
370        .await
371        .map_err(|e| format!("raw info_get_transaction body: {e}"))?;
372    if let Some(err) = value.get("error") {
373        return Err(format!("raw info_get_transaction rpc error: {err}"));
374    }
375    value
376        .get("result")
377        .cloned()
378        .ok_or_else(|| "raw info_get_transaction missing result".to_string())
379}

381fn extract_execution_error(json: &Value) -> Option<String> {
382    let candidates = [
383        json.pointer("/execution_info/execution_result/Version2/error_message"),
384        json.pointer("/execution_info/execution_result/Version1/error_message"),
385        json.pointer("/execution_info/execution_result/Failure/error_message"),
386        json.pointer("/execution_result/Version2/error_message"),
387        json.pointer("/execution_result/Version1/error_message"),
388        json.pointer("/execution_result/Failure/error_message"),
389        json.pointer("/execution_info/execution_result/error_message"),
390        json.pointer("/execution_info/execution_result/failure/error_message"),
391        json.pointer("/execution_result/failure/error_message"),
392    ];
393    for c in candidates.into_iter().flatten() {
394        if let Some(s) = c.as_str() {
395            if !s.is_empty() {
396                return Some(s.to_string());
397            }
398        }
399    }
400    let text = json.to_string();
401    if let Some(idx) = text.find("User error:") {
402        let slice: String = text[idx..].chars().take(80).collect();
403        return Some(slice);
404    }
405    if text.contains("\"Failure\"") || text.contains("\"failure\"") {
406        if let Some(idx) = text.find("error_message") {
407            let slice: String = text[idx..].chars().take(120).collect();
408            return Some(format!("execution Failure: {slice}"));
409        }
410        return Some("execution Failure (see transaction result)".into());
411    }
412    None
413}
