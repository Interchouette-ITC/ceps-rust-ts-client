1//! Named-key and dictionary queries against the bound contract.
2
3use super::CEPClient;
4use crate::error::{CEPError, Result};
5use casper_rust_wasm_sdk::helpers::contract_hash_key_for_global_state;
6use casper_rust_wasm_sdk::rpcs::get_dictionary_item::DictionaryItemInput;
7use casper_rust_wasm_sdk::rpcs::query_global_state::PathIdentifierInput;
8use casper_rust_wasm_sdk::types::identifier::dictionary_item_identifier::DictionaryItemIdentifier;
9use serde_json::Value;
10
11pub(super) async fn query_contract_key(core: &CEPClient, path: &[&str]) -> Result<Value> {
12    let target = core.require_target()?;
13    let key = contract_hash_key_for_global_state(&target.query_key());
14    // Pass the hash as a string so SDK falls back to classic `hash-…` global-state
15    // queries when addressable entities are disabled (EntityIdentifier rejects `hash-`).
16    let path_input = if path.len() == 1 {
17        PathIdentifierInput::String(path[0].to_string())
18    } else {
19        PathIdentifierInput::String(path.join("/"))
20    };
21    let response = core
22        .sdk()
23        .query_contract_key(
24            None,
25            Some(key),
26            path_input,
27            None,
28            Some(core.verbosity()),
29            Some(core.rpc_url().to_string()),
30        )
31        .await?;
32    serde_json::to_value(&response.result)
33        .map_err(|e| CEPError::Decode(format!("query_contract_key: {e}")))
34}

36pub(super) async fn query_dictionary(
37    core: &CEPClient,
38    dictionary_name: &str,
39    item_key: &str,
40) -> Result<Value> {
41    let target = core.require_target()?;
42    let key = contract_hash_key_for_global_state(&target.query_key());
43    let identifier =
44        DictionaryItemIdentifier::new_from_contract_info(&key, dictionary_name, item_key)
45            .map_err(|e| CEPError::InvalidHash(format!("dictionary item: {e}")))?;
46    let input = DictionaryItemInput::Identifier(identifier);
47    let response = core
48        .sdk()
49        .query_contract_dict(
50            input,
51            None::<&str>,
52            Some(core.verbosity()),
53            Some(core.rpc_url().to_string()),
54        )
55        .await?;
56    serde_json::to_value(&response.result)
57        .map_err(|e| CEPError::Decode(format!("query_dictionary: {e}")))
58}

60pub(super) async fn get_account_named_key(
61    core: &CEPClient,
62    account_identifier: &str,
63    named_key: &str,
64) -> Result<String> {
65    #[allow(deprecated)]
66    let response = core
67        .sdk()
68        .get_account(
69            None,
70            Some(account_identifier.to_string()),
71            None,
72            Some(core.verbosity()),
73            Some(core.rpc_url().to_string()),
74        )
75        .await?;
76
77    let keys = response.result.account.named_keys();
78    let key = keys.get(named_key).ok_or_else(|| {
79        let names: Vec<String> = keys.names().cloned().collect();
80        CEPError::EmptyQuery(format!("account named key '{named_key}' (have: {names:?})"))
81    })?;
82    Ok(key.to_formatted_string())
83}
