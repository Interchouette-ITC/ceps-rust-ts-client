278    /// Change security lists (at least one list required).
279    pub async fn change_security(
280        &self,
281        args: &ChangeSecurityArgs,
282        tx: &TransactionParams,
283    ) -> Result<CallResult> {
284        let mut json_args_vec: Vec<JsonArg> = Vec::new();
285        if let Some(list) = &args.admin_list {
286            let keys = map_keys(list)?;
287            json_args_vec.push(arg(&sch::ep::CS_ADMIN_LIST, json!(keys)));
288        }
289        if let Some(list) = &args.minter_list {
290            let keys = map_keys(list)?;
291            json_args_vec.push(arg(&sch::ep::CS_MINTER_LIST, json!(keys)));
292        }
293        if let Some(list) = &args.none_list {
294            let keys = map_keys(list)?;
295            json_args_vec.push(arg(&sch::ep::CS_NONE_LIST, json!(keys)));
296        }
297        if json_args_vec.is_empty() {
298            return Err(CEPError::MissingArgument(
299                "change_security requires at least one list".into(),
300            ));
301        }
302        let args_json = json_args(&json_args_vec);
303        self.core
304            .call_entrypoint("change_security", tx, &args_json)
305            .await
306    }

370    pub async fn security_badge(&self, account: &str) -> Result<Option<SecurityBadge18>> {
371        let item_key = balance_dictionary_key(account)?;
372        match self
373            .core
374            .query_dictionary("security_badges", &item_key)
375            .await
376        {
377            Ok(raw) => {
378                let v = decode_u8_cl(raw)?;
379                Ok(SecurityBadge18::from_u8(v))
380            }
381            Err(CEPError::EmptyQuery(_)) | Err(CEPError::Sdk(_)) => Ok(None),
382            Err(e) => Err(e),
383        }
384    }

387fn install_args_json(args: &InstallArgs) -> Result<String> {
388    let mut v = vec![
389        arg(&sch::install::NAME, Value::String(args.name.clone())),
390        arg(&sch::install::SYMBOL, Value::String(args.symbol.clone())),
391        arg(&sch::install::DECIMALS, json!(args.decimals)),
392        arg(
393            &sch::install::TOTAL_SUPPLY,
394            Value::String(args.total_supply.clone()),
395        ),
396    ];
397    if let Some(mode) = args.events_mode {
398        v.push(arg(&sch::install::EVENTS_MODE, json!(u8::from(mode))));
399    }
400    if let Some(enable) = args.enable_mint_and_burn {
401        v.push(arg(
402            &sch::install::ENABLE_MINT_BURN,
403            json!(u8::from(enable)),
404        ));
405    }
406    if let Some(list) = &args.admin_list {
407        v.push(arg(&sch::install::ADMIN_LIST, json!(map_keys(list)?)));
408    }
409    if let Some(list) = &args.minter_list {
410        v.push(arg(&sch::install::MINTER_LIST, json!(map_keys(list)?)));
411    }
412    Ok(json_args(&v))
413}

419fn decode_string_cl(value: Value) -> Result<String> {
420    // Prefer parsed CLValue JSON shapes from query_global_state.
421    if let Some(s) = value
422        .pointer("/stored_value/CLValue/parsed")
423        .and_then(|v| v.as_str())
424    {
425        return Ok(s.to_string());
426    }
427    if let Some(s) = value.pointer("/CLValue/parsed").and_then(|v| v.as_str()) {
428        return Ok(s.to_string());
429    }
430    if let Some(s) = value.as_str() {
431        return Ok(s.to_string());
432    }
433    Err(CEPError::Decode(format!(
434        "expected string CLValue, got {value}"
435    )))
436}
437
438fn decode_u8_cl(value: Value) -> Result<u8> {
439    let parsed = value
440        .pointer("/stored_value/CLValue/parsed")
441        .or_else(|| value.pointer("/CLValue/parsed"))
442        .cloned()
443        .unwrap_or(value);
444    match parsed {
445        Value::Number(n) => n
446            .as_u64()
447            .and_then(|v| u8::try_from(v).ok())
448            .ok_or_else(|| CEPError::Decode(format!("invalid u8: {n}"))),
449        Value::String(s) => s
450            .parse()
451            .map_err(|e| CEPError::Decode(format!("invalid u8 string: {e}"))),
452        other => Err(CEPError::Decode(format!("expected u8, got {other}"))),
453    }
454}
455
456fn decode_u256_cl(value: Value) -> Result<String> {
457    let parsed = value
458        .pointer("/stored_value/CLValue/parsed")
459        .or_else(|| value.pointer("/CLValue/parsed"))
460        .cloned()
461        .unwrap_or(value);
462    match parsed {
463        Value::String(s) => Ok(s),
464        Value::Number(n) => Ok(n.to_string()),
465        other => Err(CEPError::Decode(format!("expected U256, got {other}"))),
466    }
467}
