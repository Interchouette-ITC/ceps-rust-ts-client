127    /// Upgrade an existing CEP-78 package (same installer WASM).
128    pub async fn upgrade(
129        &self,
130        args: &UpgradeArgs,
131        wasm: &[u8],
132        tx: &TransactionParams,
133    ) -> Result<CallResult> {
134        let mut v = vec![arg(
135            &sch::ep::UPGRADE_COLLECTION_NAME,
136            Value::String(args.collection_name.clone()),
137        )];
138        if let Some(supply) = args.total_token_supply {
139            v.push(arg(&sch::ep::UPGRADE_TOTAL_TOKEN_SUPPLY, json!(supply)));
140        }
141        if let Some(mode) = args.events_mode {
142            v.push(arg(&sch::ep::UPGRADE_EVENTS_MODE, json!(u8::from(mode))));
143        }
144        if let Some(b) = args.acl_package_mode {
145            v.push(arg(&sch::ep::UPGRADE_ACL_PACKAGE_MODE, json!(b)));
146        }
147        if let Some(b) = args.package_operator_mode {
148            v.push(arg(&sch::ep::UPGRADE_PACKAGE_OPERATOR_MODE, json!(b)));
149        }
150        if let Some(b) = args.operator_burn_mode {
151            v.push(arg(&sch::ep::UPGRADE_OPERATOR_BURN_MODE, json!(b)));
152        }
153        self.core.install_wasm(wasm, tx, &json_args(&v)).await
154    }

344    /// Update collection variables.
345    pub async fn set_variables(
346        &self,
347        args: &SetVariablesArgs,
348        tx: &TransactionParams,
349    ) -> Result<CallResult> {
350        let mut v: Vec<JsonArg> = Vec::new();
351        if let Some(b) = args.allow_minting {
352            v.push(arg(&sch::ep::SV_ALLOW_MINTING, json!(b)));
353        }
354        if let Some(list) = &args.acl_whitelist {
355            let keys: Result<Vec<_>> = list.iter().map(|k| prefixed_key(k)).collect();
356            v.push(arg(&sch::ep::SV_ACL_WHITELIST, json!(keys?)));
357        }
358        if let Some(b) = args.acl_package_mode {
359            v.push(arg(&sch::ep::SV_ACL_PACKAGE_MODE, json!(b)));
360        }
361        if let Some(b) = args.package_operator_mode {
362            v.push(arg(&sch::ep::SV_PACKAGE_OPERATOR_MODE, json!(b)));
363        }
364        if let Some(b) = args.operator_burn_mode {
365            v.push(arg(&sch::ep::SV_OPERATOR_BURN_MODE, json!(b)));
366        }
367        if v.is_empty() {
368            return Err(CEPError::MissingArgument(
369                "set_variables requires at least one field".into(),
370            ));
371        }
372        self.core
373            .call_entrypoint("set_variables", tx, &json_args(&v))
374            .await
375    }

390        self.core
391            .call_session(session_wasm, tx, &json_args(&v))
392            .await
393    }

666            &sch::install::TOTAL_TOKEN_SUPPLY,
667            json!(args.total_token_supply),
668        ),
669        arg(
670            &sch::install::OWNERSHIP_MODE,
671            json!(u8::from(args.ownership_mode)),
672        ),
673        arg(
674            &sch::install::NFT_METADATA_KIND,
675            json!(u8::from(args.nft_metadata_kind)),
676        ),
677        arg(
678            &sch::install::IDENTIFIER_MODE,
679            json!(u8::from(args.identifier_mode)),
680        ),
681        arg(
682            &sch::install::METADATA_MUTABILITY,
683            json!(u8::from(args.metadata_mutability)),
684        ),
685    ];
686    if let Some(kind) = args.nft_kind {
687        v.push(arg(&sch::install::NFT_KIND, json!(u8::from(kind))));
688    }
689    if let Some(schema) = &args.json_schema {
690        v.push(arg(
691            &sch::install::JSON_SCHEMA,
692            Value::String(schema.clone()),
693        ));
694    }
695    if let Some(mode) = args.minting_mode {
696        v.push(arg(&sch::install::MINTING_MODE, json!(u8::from(mode))));
697    }
698    if let Some(b) = args.allow_minting {
699        v.push(arg(&sch::install::ALLOW_MINTING, json!(b)));
700    }
701    if let Some(b) = args.operator_burn_mode {
702        v.push(arg(&sch::install::OPERATOR_BURN_MODE, json!(b)));
703    }
704    if let Some(b) = args.package_operator_mode {
705        v.push(arg(&sch::install::PACKAGE_OPERATOR_MODE, json!(b)));
706    }
707    if let Some(mode) = args.whitelist_mode {
708        v.push(arg(&sch::install::WHITELIST_MODE, json!(u8::from(mode))));
709    }
710    if let Some(mode) = args.holder_mode {
711        v.push(arg(&sch::install::HOLDER_MODE, json!(u8::from(mode))));
712    }
713    if let Some(b) = args.acl_package_mode {
714        v.push(arg(&sch::install::ACL_PACKAGE_MODE, json!(b)));
715    }
716    if let Some(list) = &args.acl_whitelist {
717        let keys: Result<Vec<_>> = list.iter().map(|k| prefixed_key(k)).collect();
718        v.push(arg(&sch::install::ACL_WHITELIST, json!(keys?)));
719    }
720    if let Some(mode) = args.burn_mode {
721        v.push(arg(&sch::install::BURN_MODE, json!(u8::from(mode))));
722    }
723    if let Some(mode) = args.owner_reverse_lookup_mode {
724        v.push(arg(
725            &sch::install::OWNER_REVERSE_LOOKUP_MODE,
726            json!(u8::from(mode)),
727        ));
728    }
729    if let Some(mode) = args.named_key_convention {
730        v.push(arg(
731            &sch::install::NAMED_KEY_CONVENTION,
732            json!(u8::from(mode)),
733        ));
734    }
735    if let Some(name) = &args.access_key_name {
736        v.push(arg(
737            &sch::install::ACCESS_KEY_NAME,
738            Value::String(name.clone()),
739        ));
740    }
741    if let Some(name) = &args.hash_key_name {
742        v.push(arg(
743            &sch::install::HASH_KEY_NAME,
744            Value::String(name.clone()),
745        ));
746    }
747    if let Some(mode) = args.events_mode {
748        v.push(arg(&sch::install::EVENTS_MODE, json!(u8::from(mode))));
749    }
750    if let Some(filter) = &args.transfer_filter_contract {
751        v.push(arg(
752            &sch::install::TRANSFER_FILTER_CONTRACT,
753            Value::String(prefixed_key(filter)?),
754        ));
755    }
756    if matches!(
757        args.named_key_convention,
758        Some(NamedKeyConventionMode::V1_0Custom)
759    ) && (args.access_key_name.is_none() || args.hash_key_name.is_none())
760    {
761        return Err(CEPError::MissingArgument(
762            "V1_0Custom requires access_key_name and hash_key_name".into(),
763        ));
764    }
765    Ok(json_args(&v))
766}

784fn decode_string_cl(value: Value) -> Result<String> {
785    if let Some(s) = value
786        .pointer("/stored_value/CLValue/parsed")
787        .and_then(|v| v.as_str())
788    {
789        return Ok(s.to_string());
790    }
791    if let Some(s) = value.pointer("/CLValue/parsed").and_then(|v| v.as_str()) {
792        return Ok(s.to_string());
793    }
794    Err(CEPError::Decode(format!("expected string, got {value}")))
795}

797fn decode_u8_cl(value: Value) -> Result<u8> {
798    let parsed = value
799        .pointer("/stored_value/CLValue/parsed")
800        .or_else(|| value.pointer("/CLValue/parsed"))
801        .cloned()
802        .unwrap_or(value);
803    match parsed {
804        Value::Number(n) => n
805            .as_u64()
806            .map(|v| v as u8)
807            .ok_or_else(|| CEPError::Decode(format!("expected u8, got {n}"))),
808        Value::String(s) => s
809            .parse()
810            .map_err(|e| CEPError::Decode(format!("expected u8: {e}"))),
811        other => Err(CEPError::Decode(format!("expected u8, got {other}"))),
812    }
813}

815fn decode_u64_cl(value: Value) -> Result<u64> {
816    let parsed = value
817        .pointer("/stored_value/CLValue/parsed")
818        .or_else(|| value.pointer("/CLValue/parsed"))
819        .cloned()
820        .unwrap_or(value);
821    match parsed {
822        Value::Number(n) => n
823            .as_u64()
824            .ok_or_else(|| CEPError::Decode(format!("expected u64, got {n}"))),
825        Value::String(s) => s
826            .parse()
827            .map_err(|e| CEPError::Decode(format!("expected u64: {e}"))),
828        other => Err(CEPError::Decode(format!("expected u64, got {other}"))),
829    }
830}

832fn decode_u64_string(value: Value) -> Result<String> {
833    Ok(decode_u64_cl(value)?.to_string())
834}

836fn decode_bool_cl(value: Value) -> Result<bool> {
837    let parsed = value
838        .pointer("/stored_value/CLValue/parsed")
839        .or_else(|| value.pointer("/CLValue/parsed"))
840        .cloned()
841        .unwrap_or(value);
842    match parsed {
843        Value::Bool(b) => Ok(b),
844        Value::Number(n) => Ok(n.as_u64().unwrap_or(0) != 0),
845        Value::String(s) => match s.to_ascii_lowercase().as_str() {
846            "true" | "1" => Ok(true),
847            "false" | "0" => Ok(false),
848            other => Err(CEPError::Decode(format!(
849                "expected bool string, got {other}"
850            ))),
851        },
852        other => Err(CEPError::Decode(format!("expected bool, got {other}"))),
853    }
854}

856async fn mode_u8<T, F>(client: &CEP78Client, named_key: &str, map: F) -> Result<T>
857where
858    F: FnOnce(u8) -> Option<T>,
859{
860    let v = decode_u8_cl(client.core.query_contract_key(&[named_key]).await?)?;
861    map(v).ok_or_else(|| CEPError::Decode(format!("unknown {named_key} {v}")))
862}
863
864fn decode_key_cl(value: Value) -> Result<String> {
865    if let Some(s) = value
866        .pointer("/stored_value/CLValue/parsed")
867        .and_then(|v| v.as_str())
868    {
869        return Ok(s.to_string());
870    }
871    if let Some(obj) = value.pointer("/stored_value/CLValue/parsed") {
872        if let Some(s) = obj.as_str() {
873            return Ok(s.to_string());
874        }
875        return Ok(obj.to_string());
876    }
877    if let Some(s) = value.pointer("/CLValue/parsed").and_then(|v| v.as_str()) {
878        return Ok(s.to_string());
879    }
880    Err(CEPError::Decode(format!("expected Key, got {value}")))
881}
