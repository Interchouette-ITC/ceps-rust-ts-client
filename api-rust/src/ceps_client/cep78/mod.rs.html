127    /// Upgrade an existing CEP-78 package (same installer WASM).
128    pub async fn upgrade(
129        &self,
130        args: &UpgradeArgs,
131        wasm: &[u8],
132        tx: &TransactionParams,
133    ) -> Result<CallResult> {
134        let mut v = vec![string_arg("collection_name", &args.collection_name)];
135        if let Some(supply) = args.total_token_supply {
136            v.push(u64_arg("total_token_supply", supply));
137        }
138        if let Some(mode) = args.events_mode {
139            v.push(u8_arg("events_mode", mode.into()));
140        }
141        if let Some(b) = args.acl_package_mode {
142            v.push(bool_arg("acl_package_mode", b));
143        }
144        if let Some(b) = args.package_operator_mode {
145            v.push(bool_arg("package_operator_mode", b));
146        }
147        if let Some(b) = args.operator_burn_mode {
148            v.push(bool_arg("operator_burn_mode", b));
149        }
150        self.core.install_wasm(wasm, tx, &json_args(&v)).await
151    }

311    /// Update collection variables.
312    pub async fn set_variables(
313        &self,
314        args: &SetVariablesArgs,
315        tx: &TransactionParams,
316    ) -> Result<CallResult> {
317        let mut v: Vec<JsonArg> = Vec::new();
318        if let Some(b) = args.allow_minting {
319            v.push(bool_arg("allow_minting", b));
320        }
321        if let Some(list) = &args.acl_whitelist {
322            let keys: Result<Vec<_>> = list.iter().map(|k| prefixed_key(k)).collect();
323            v.push(key_list_arg("acl_whitelist", &keys?));
324        }
325        if let Some(b) = args.acl_package_mode {
326            v.push(bool_arg("acl_package_mode", b));
327        }
328        if let Some(b) = args.package_operator_mode {
329            v.push(bool_arg("package_operator_mode", b));
330        }
331        if let Some(b) = args.operator_burn_mode {
332            v.push(bool_arg("operator_burn_mode", b));
333        }
334        if v.is_empty() {
335            return Err(CEPError::MissingArgument(
336                "set_variables requires at least one field".into(),
337            ));
338        }
339        self.core
340            .call_entrypoint("set_variables", tx, &json_args(&v))
341            .await
342    }

622fn install_args_json(args: &InstallArgs) -> Result<String> {
623    let mut v = vec![
624        string_arg("collection_name", &args.collection_name),
625        string_arg("collection_symbol", &args.collection_symbol),
626        u64_arg("total_token_supply", args.total_token_supply),
627        u8_arg("ownership_mode", args.ownership_mode.into()),
628        u8_arg("nft_metadata_kind", args.nft_metadata_kind.into()),
629        u8_arg("identifier_mode", args.identifier_mode.into()),
630        u8_arg("metadata_mutability", args.metadata_mutability.into()),
631    ];
632    if let Some(kind) = args.nft_kind {
633        v.push(u8_arg("nft_kind", kind.into()));
634    }
635    if let Some(schema) = &args.json_schema {
636        v.push(string_arg("json_schema", schema));
637    }
638    if let Some(mode) = args.minting_mode {
639        v.push(u8_arg("minting_mode", mode.into()));
640    }
641    if let Some(b) = args.allow_minting {
642        v.push(bool_arg("allow_minting", b));
643    }
644    if let Some(b) = args.operator_burn_mode {
645        v.push(bool_arg("operator_burn_mode", b));
646    }
647    if let Some(b) = args.package_operator_mode {
648        v.push(bool_arg("package_operator_mode", b));
649    }
650    if let Some(mode) = args.whitelist_mode {
651        v.push(u8_arg("whitelist_mode", mode.into()));
652    }
653    if let Some(mode) = args.holder_mode {
654        v.push(u8_arg("holder_mode", mode.into()));
655    }
656    if let Some(b) = args.acl_package_mode {
657        v.push(bool_arg("acl_package_mode", b));
658    }
659    if let Some(list) = &args.acl_whitelist {
660        let keys: Result<Vec<_>> = list.iter().map(|k| prefixed_key(k)).collect();
661        v.push(key_list_arg("acl_whitelist", &keys?));
662    }
663    if let Some(mode) = args.burn_mode {
664        v.push(u8_arg("burn_mode", mode.into()));
665    }
666    if let Some(mode) = args.owner_reverse_lookup_mode {
667        v.push(u8_arg("owner_reverse_lookup_mode", mode.into()));
668    }
669    if let Some(mode) = args.named_key_convention {
670        v.push(u8_arg("named_key_convention", mode.into()));
671    }
672    if let Some(name) = &args.access_key_name {
673        v.push(string_arg("access_key_name", name));
674    }
675    if let Some(name) = &args.hash_key_name {
676        v.push(string_arg("hash_key_name", name));
677    }
678    if let Some(mode) = args.events_mode {
679        v.push(u8_arg("events_mode", mode.into()));
680    }
681    if let Some(filter) = &args.transfer_filter_contract {
682        v.push(key_arg("transfer_filter_contract", &prefixed_key(filter)?));
683    }
684    if matches!(
685        args.named_key_convention,
686        Some(NamedKeyConventionMode::V1_0Custom)
687    ) && (args.access_key_name.is_none() || args.hash_key_name.is_none())
688    {
689        return Err(CEPError::MissingArgument(
690            "V1_0Custom requires access_key_name and hash_key_name".into(),
691        ));
692    }
693    Ok(json_args(&v))
694}

710fn decode_string_cl(value: Value) -> Result<String> {
711    if let Some(s) = value
712        .pointer("/stored_value/CLValue/parsed")
713        .and_then(|v| v.as_str())
714    {
715        return Ok(s.to_string());
716    }
717    if let Some(s) = value.pointer("/CLValue/parsed").and_then(|v| v.as_str()) {
718        return Ok(s.to_string());
719    }
720    Err(CEPError::Decode(format!("expected string, got {value}")))
721}

723fn decode_u8_cl(value: Value) -> Result<u8> {
724    let parsed = value
725        .pointer("/stored_value/CLValue/parsed")
726        .or_else(|| value.pointer("/CLValue/parsed"))
727        .cloned()
728        .unwrap_or(value);
729    match parsed {
730        Value::Number(n) => n
731            .as_u64()
732            .map(|v| v as u8)
733            .ok_or_else(|| CEPError::Decode(format!("expected u8, got {n}"))),
734        Value::String(s) => s
735            .parse()
736            .map_err(|e| CEPError::Decode(format!("expected u8: {e}"))),
737        other => Err(CEPError::Decode(format!("expected u8, got {other}"))),
738    }
739}

741fn decode_u64_cl(value: Value) -> Result<u64> {
742    let parsed = value
743        .pointer("/stored_value/CLValue/parsed")
744        .or_else(|| value.pointer("/CLValue/parsed"))
745        .cloned()
746        .unwrap_or(value);
747    match parsed {
748        Value::Number(n) => n
749            .as_u64()
750            .ok_or_else(|| CEPError::Decode(format!("expected u64, got {n}"))),
751        Value::String(s) => s
752            .parse()
753            .map_err(|e| CEPError::Decode(format!("expected u64: {e}"))),
754        other => Err(CEPError::Decode(format!("expected u64, got {other}"))),
755    }
756}
757
758fn decode_u64_string(value: Value) -> Result<String> {
759    Ok(decode_u64_cl(value)?.to_string())
760}

762fn decode_bool_cl(value: Value) -> Result<bool> {
763    let parsed = value
764        .pointer("/stored_value/CLValue/parsed")
765        .or_else(|| value.pointer("/CLValue/parsed"))
766        .cloned()
767        .unwrap_or(value);
768    match parsed {
769        Value::Bool(b) => Ok(b),
770        Value::Number(n) => Ok(n.as_u64().unwrap_or(0) != 0),
771        Value::String(s) => match s.to_ascii_lowercase().as_str() {
772            "true" | "1" => Ok(true),
773            "false" | "0" => Ok(false),
774            other => Err(CEPError::Decode(format!(
775                "expected bool string, got {other}"
776            ))),
777        },
778        other => Err(CEPError::Decode(format!("expected bool, got {other}"))),
779    }
780}

782async fn mode_u8<T, F>(client: &CEP78Client, named_key: &str, map: F) -> Result<T>
783where
784    F: FnOnce(u8) -> Option<T>,
785{
786    let v = decode_u8_cl(client.core.query_contract_key(&[named_key]).await?)?;
787    map(v).ok_or_else(|| CEPError::Decode(format!("unknown {named_key} {v}")))
788}

790fn decode_key_cl(value: Value) -> Result<String> {
791    if let Some(s) = value
792        .pointer("/stored_value/CLValue/parsed")
793        .and_then(|v| v.as_str())
794    {
795        return Ok(s.to_string());
796    }
797    if let Some(obj) = value.pointer("/stored_value/CLValue/parsed") {
798        if let Some(s) = obj.as_str() {
799            return Ok(s.to_string());
800        }
801        return Ok(obj.to_string());
802    }
803    if let Some(s) = value.pointer("/CLValue/parsed").and_then(|v| v.as_str()) {
804        return Ok(s.to_string());
805    }
806    Err(CEPError::Decode(format!("expected Key, got {value}")))
807}
