3use serde::ser::{SerializeStruct, Serializer};
4use serde::Serialize;
5use serde_json::Value;
6use std::collections::BTreeMap;

8/// One runtime / install argument descriptor (`session_args_json` CLType shape).
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub struct ArgField {
11    /// Argument name as expected by the contract / installer.
12    pub name: &'static str,
13    /// Whether the helper omits the arg when unset.
14    pub optional: bool,
15    /// Static JSON fragment for the CLType (`"String"`, `{"List":"Key"}`, …).
16    type_json: &'static str,
17}
18
19impl ArgField {
20    /// Create a field descriptor.
21    pub const fn new(name: &'static str, type_json: &'static str, optional: bool) -> Self {
22        Self {
23            name,
24            optional,
25            type_json,
26        }
27    }
28
29    /// Required field.
30    pub const fn req(name: &'static str, type_json: &'static str) -> Self {
31        Self::new(name, type_json, false)
32    }
33
34    /// Optional field.
35    pub const fn opt(name: &'static str, type_json: &'static str) -> Self {
36        Self::new(name, type_json, true)
37    }
38
39    /// CLType as `serde_json::Value` (casper-client `session_args_json` shape).
40    pub fn cl_type(&self) -> Value {
41        serde_json::from_str(self.type_json).unwrap_or_else(|e| {
42            panic!(
43                "invalid static CLType JSON for {}: {e}: {}",
44                self.name, self.type_json
45            )
46        })
47    }
48}
49
50impl Serialize for ArgField {
51    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
52        let mut state = serializer.serialize_struct("ArgField", 3)?;
53        state.serialize_field("name", self.name)?;
54        state.serialize_field("type", &self.cl_type())?;
55        state.serialize_field("optional", &self.optional)?;
56        state.end()
57    }
58}

60/// Full schema for one CEP (install + mutate entrypoints).
61#[derive(Debug, Clone, Serialize)]
62pub struct CepSchema {
63    /// Stable id (`cep18`, `cep78`, `cep85`, `cep95`).
64    pub cep: &'static str,
65    /// Installer session args.
66    pub install: Vec<ArgField>,
67    /// Entrypoint name → runtime args.
68    pub entrypoints: BTreeMap<&'static str, Vec<ArgField>>,
69}
70
71impl CepSchema {
72    pub(crate) fn from_parts(
73        cep: &'static str,
74        install: &'static [ArgField],
75        entrypoints: &'static [(&'static str, &'static [ArgField])],
76    ) -> Self {
77        let mut map = BTreeMap::new();
78        for (name, fields) in entrypoints {
79            map.insert(*name, fields.to_vec());
80        }
81        Self {
82            cep,
83            install: install.to_vec(),
84            entrypoints: map,
85        }
86    }
87}
